func corpusOf(stations []gen.Station, rows int, seed uint64) *corpus {
	var buf bytes.Buffer
	buf.Grow(rows * 16)
	if _, err := gen.Write(&buf, stations, int64(rows), 10.0, seed); err != nil {
		panic(err)

	padded := make([]byte, len(b), len(b)+16)

	return &corpus{bytes: padded, rows: bytes.Count(b, []byte{'\n'})}

// TestPrefixCollisions counts, for both key sets, how many distinct names share an 8- and a 16-byte prefix.
// A prefix hash is only usable if a full-key compare resolves the collisions cheaply, so the number that matters is how many names land on the busiest prefix.
func TestPrefixCollisions(t *testing.T) {
	for _, ks := range []struct {
		name     string
		stations []gen.Station
	}{{"413-official", gen.Official413()}, {"10k-synthetic", gen.Synthetic10k()}} {
		names := make([]string, 0, len(ks.stations))
		for _, s := range ks.stations {
			names = append(names, s.Name)
		}
		for _, w := range []int{8, 16} {
			groups := map[string][]string{}
			for _, n := range names {
				k := n
				if len(k) > w {
					k = k[:w]
				}
				groups[k] = append(groups[k], n)
			}
			worst, colliding, groupsWithCollisions := 0, 0, 0
			var worstKey string
			for k, g := range groups {
				if len(g) > 1 {
					groupsWithCollisions++
					colliding += len(g)
				}
				if len(g) > worst {
					worst, worstKey = len(g), k
				}
			}
			over := 0
			for _, n := range names {
				if len(n) > w {
					over++
				}
			}
			t.Logf("%s w=%d: names=%d distinct_prefixes=%d names_over_w=%d (%.1f%%) colliding_groups=%d colliding_names=%d worst_group=%d (%q)",
				ks.name, w, len(names), len(groups), over, 100*float64(over)/float64(len(names)), groupsWithCollisions, colliding, worst, worstKey)
		}
	}
}

// ---------- the three table shapes, one scan for all of them ----------

type acc struct {
	min, max int32
	sum      int64
	count    int32
}

const (
	fnvOffset = 14695981039346656037
	fnvPrime  = 1099511628211
	tableBits = 14
	tableSize = 1 << tableBits
)

type entry struct {
	key []byte
	a   acc
}

// probeTable is linear-probed open addressing: power-of-two buckets, key bytes copied on insert, collisions resolved by a full compare.
type probeTable struct {
	e    []entry
	mask uint64
	size int
}

func newProbeTable(bits int) *probeTable {
	return &probeTable{e: make([]entry, 1<<bits), mask: 1<<bits - 1}
}

func (t *probeTable) update(h uint64, name []byte, temp int32) {
	i := int(h & t.mask)
	for {
		if t.e[i].key == nil {
			k := make([]byte, len(name))
			copy(k, name)
			t.e[i] = entry{key: k, a: acc{min: temp, max: temp, sum: int64(temp), count: 1}}
			t.size++
			return
		}
		if bytes.Equal(t.e[i].key, name) {
			a := &t.e[i].a
			a.min = min(a.min, temp)
			a.max = max(a.max, temp)
			a.sum += int64(temp)
			a.count++
			return
		}
		i = (i + 1) & int(t.mask)
	}
}

func fnvOf(name []byte) uint64 {
	h := uint64(fnvOffset)
	for _, c := range name {
		h ^= uint64(c)
		h *= fnvPrime
	}
	return h
}

// prefixHashOf hashes at most the first 8 bytes of the name, which is what r10 in the Go corpus does; the table's full compare is what keeps it correct.
func prefixHashOf(name []byte) uint64 {
	var w uint64
	if len(name) >= 8 {
		w = binary.LittleEndian.Uint64(name)
	} else {
		for i := len(name) - 1; i >= 0; i-- {
			w = w<<8 | uint64(name[i])
		}
	}
	return (w ^ w>>29) * 0x9E3779B97F4A7C15
}

func parseTemp(b []byte) (int32, int) {
	i := 0
	neg := false
	if b[i] == '-' {
		neg = true
		i++
	}
	t := int32(b[i] - '0')
	i++
	if b[i] != '.' {
		t = t*10 + int32(b[i]-'0')
		i++
	}
	i++
	t = t*10 + int32(b[i]-'0')
	i += 2
	if neg {
		t = -t
	}
	return t, i
}

// TestTablesAgree checks the three table shapes and the reference all produce the same aggregate, so the benchmark above is pricing correct work.
func TestTablesAgree(t *testing.T) {
	c := corpus413()
	want, err := gen.Aggregate(bytes.NewReader(c.bytes))
	if err != nil {
		t.Fatal(err)
	}
	for _, tc := range []struct {
		label string
		hash  func([]byte) uint64
	}{{"fnv", fnvOf}, {"prefix8", prefixHashOf}} {
		tab := newProbeTable(tableBits)
		for pos := 0; pos < len(c.bytes); {
			sep := bytes.IndexByte(c.bytes[pos:], ';')
			name := c.bytes[pos : pos+sep]
			temp, n := parseTemp(c.bytes[pos+sep+1:])
			tab.update(tc.hash(name), name, temp)
			pos += sep + 1 + n
		}
		if tab.size != len(want) {
			t.Fatalf("%s: %d stations, want %d", tc.label, tab.size, len(want))
		}
		for _, e := range tab.e {
			if e.key == nil {
				continue
			}
			w := want[string(e.key)]
			if w == nil {
				t.Fatalf("%s: %q not in reference", tc.label, e.key)
			}
			if int64(e.a.min) != int64(w.Min) || int64(e.a.max) != int64(w.Max) || e.a.sum != int64(w.Sum) || int64(e.a.count) != w.Count {
				t.Fatalf("%s: %q got %+v want min=%d max=%d sum=%d count=%d", tc.label, e.key, e.a, w.Min, w.Max, w.Sum, w.Count)
			}
		}
	}
}

// ---------- probe: slice-indexed SWAR against an unsafe.Pointer walk ----------

const (
	semiBroadcast = 0x3B3B3B3B3B3B3B3B
	swarLow       = 0x0101010101010101
	swarHigh      = 0x8080808080808080
)

func swarMatch(w uint64) uint64 {
	x := w ^ semiBroadcast
	return (x - swarLow) & ^x & swarHigh
}

func swarIndexSlice(b []byte) int {
	i := 0
	for ; i+8 <= len(b); i += 8 {
		if m := swarMatch(binary.LittleEndian.Uint64(b[i:])); m != 0 {
			return i + bits.TrailingZeros64(m)>>3
		}
	}
	for ; i < len(b); i++ {
		if b[i] == ';' {
			return i
		}
	}
	return -1
}

// swarIndexUnsafe is swarIndexSlice with the same arithmetic and an unsafe.Pointer walk instead of a slice index, which is the comparison the recon owes: does unsafe buy anything once the bounds check is already eliminated?
func swarIndexUnsafe(b []byte) int {
	if len(b) == 0 {
		return -1
	}
	p := unsafe.Pointer(unsafe.SliceData(b))
	i := 0
	for ; i+8 <= len(b); i += 8 {
		if m := swarMatch(*(*uint64)(unsafe.Add(p, i))); m != 0 {
			return i + bits.TrailingZeros64(m)>>3
		}
	}
	for ; i < len(b); i++ {
		if *(*byte)(unsafe.Add(p, i)) == ';' {
			return i
		}
	}
	return -1
}

// swarIndexResliced is the stdlib's shape: guard on len, load from the front, advance the slice.
// The prover sees the length guard immediately before the load, which is what the indexed version denies it.
func swarIndexResliced(b []byte) int {
	rest, off := b, 0
	for len(rest) >= 8 {
		if m := swarMatch(binary.LittleEndian.Uint64(rest)); m != 0 {
			return off + bits.TrailingZeros64(m)>>3
		}
		rest = rest[8:]
		off += 8
	}
	for i, c := range rest {
		if c == ';' {
			return off + i
		}
	}
	return -1
}

		}

		b.ReportMetric(float64(b.Elapsed().Nanoseconds())/float64(b.N), "ns/call")
	})
	b.Run("go-inline", func(b *testing.B) {
		acc := uint64(0)
		for range b.N {
			acc = goAddInline(acc, 1)
		}
		runtime.KeepAlive(acc)
		b.ReportMetric(float64(b.Elapsed().Nanoseconds())/float64(b.N), "ns/call")
	})
}

func TestAsmAdd(t *testing.T) {
	if got := asmAdd(41, 1); got != 42 {
		t.Fatalf("asmAdd(41,1) = %d", got)
	}
	if fmt.Sprint(asmAdd(^uint64(0), 1)) != "0" {
		t.Fatal("asmAdd does not wrap like Go's +")
	}
}
